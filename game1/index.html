const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); const gravity = 0.7; const groundY = 380; class Player { constructor({ x, facing, hpId, mpId, color }) { this.x = x; this.y = 200; this.width = 50; this.height = 80; this.velocity = { x: 0, y: 0 }; this.facing = facing; this.hp = 100; this.mp = 0; this.hpId = hpId; this.mpId = mpId; this.color = color; this.isAttacking = false; this.attackBox = { x: 0, y: 0, width: 60, height: 40 }; } draw() { // Vẽ khối nhân vật ctx.fillStyle = this.color; ctx.fillRect(this.x, this.y, this.width, this.height); // Vẽ mắt chỉ hướng quay mặt ctx.fillStyle = '#fff'; const eyeX = this.facing === 'right' ? this.x + 35 : this.x + 5; ctx.fillRect(eyeX, this.y + 15, 10, 10); // Vẽ vùng đánh khi tấn công if (this.isAttacking) { ctx.fillStyle = 'rgba(255, 230, 0, 0.6)'; const boxX = this.facing === 'right' ? this.x + this.width : this.x - this.attackBox.width; ctx.fillRect(boxX, this.y + 20, this.attackBox.width, this.attackBox.height); } } update() { this.x += this.velocity.x; this.y += this.velocity.y; if (this.y + this.height + this.velocity.y >= groundY) { this.velocity.y = 0; this.y = groundY - this.height; } else { this.velocity.y += gravity; } if (this.x < 0) this.x = 0; if (this.x + this.width > canvas.width) this.x = canvas.width - this.width; this.draw(); } attack(boxWidth = 60, costMP = 0) { if (this.isAttacking || this.mp < costMP) return false; this.mp -= costMP; this.updateUI(); this.isAttacking = true; this.attackBox.width = boxWidth; setTimeout(() => { this.isAttacking = false; }, 200); return true; } takeDamage(amount) { this.hp = Math.max(0, this.hp - amount); this.mp = Math.min(100, this.mp + 10); this.updateUI(); } updateUI() { document.getElementById(this.hpId).style.width = this.hp + '%'; document.getElementById(this.mpId).style.width = this.mp + '%'; } } const player1 = new Player({ x: 150, facing: 'right', hpId: 'p1-hp', mpId: 'p1-mp', color: '#ef4444' }); const player2 = new Player({ x: 610, facing: 'left', hpId: 'p2-hp', mpId: 'p2-mp', color: '#3b82f6' }); const keys = { a: false, d: false, w: false, ArrowLeft: false, ArrowRight: false, ArrowUp: false }; window.addEventListener('keydown', (e) => { if (e.key === 'a' || e.key === 'A') keys.a = true; if (e.key === 'd' || e.key === 'D') keys.d = true; if (e.key === 'w' || e.key === 'W') keys.w = true; if (e.key === 'u' || e.key === 'U') { if (player1.attack(60, 0)) checkHit(player1, player2, 10); } if (e.key === 'ArrowLeft') keys.ArrowLeft = true; if (e.key === 'ArrowRight') keys.ArrowRight = true; if (e.key === 'ArrowUp') keys.ArrowUp = true; if (e.key === '1') { if (player2.attack(60, 0)) checkHit(player2, player1, 10); } }); window.addEventListener('keyup', (e) => { if (e.key === 'a' || e.key === 'A') keys.a = false; if (e.key === 'd' || e.key === 'D') keys.d = false; if (e.key === 'w' || e.key === 'W') keys.w = false; if (e.key === 'ArrowLeft') keys.ArrowLeft = false; if (e.key === 'ArrowRight') keys.ArrowRight = false; if (e.key === 'ArrowUp') keys.ArrowUp = false; }); function checkHit(attacker, defender, damage) { const boxX = attacker.facing === 'right' ? attacker.x + attacker.width : attacker.x - attacker.attackBox.width; if (boxX < defender.x + defender.width && boxX + attacker.attackBox.width > defender.x) { defender.takeDamage(damage); } } function animate() { requestAnimationFrame(animate); ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = '#334155'; ctx.fillRect(0, groundY, canvas.width, canvas.height - groundY); player1.velocity.x = keys.a ? -4 : (keys.d ? 4 : 0); if (keys.a) player1.facing = 'left'; if (keys.d) player1.facing = 'right'; if (keys.w && player1.y + player1.height >= groundY) player1.velocity.y = -14; player2.velocity.x = keys.ArrowLeft ? -4 : (keys.ArrowRight ? 4 : 0); if (keys.ArrowLeft) player2.facing = 'left'; if (keys.ArrowRight) player2.facing = 'right'; if (keys.ArrowUp && player2.y + player2.height >= groundY) player2.velocity.y = -14; player1.update(); player2.update(); } animate();
