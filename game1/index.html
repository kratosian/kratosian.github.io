// ============================================================ // NHÂN VẬT VẼ BẰNG CODE - KHÔNG CẦN ẢNH // ============================================================ function drawCharacter(ctx, hero, x, y, w, h, facing, state, frame, attacking, hurtTimer) { ctx.save(); // -------------------------------------------------------- // Hiệu ứng nhấp nháy khi bị đánh // -------------------------------------------------------- if (hurtTimer > 0 && hurtTimer % 4 < 2) { ctx.globalAlpha = 0.45; } // -------------------------------------------------------- // Lật nhân vật // -------------------------------------------------------- if (facing === "left") { ctx.translate(x + w, 0); ctx.scale(-1, 1); x = 0; } // -------------------------------------------------------- // Animation // -------------------------------------------------------- const t = performance.now() / 120; let bob = 0; if (state === "idle") { bob = Math.sin(t) * 2; } if (state === "walk") { bob = Math.sin(t * 2) * 3; } if (state === "jump") { bob = -3; } let attackMove = attacking ? Math.sin(t * 2) * 5 : 0; y += bob; // ======================================================== // VÕ SĨ // ======================================================== if (hero === "vosi") { // Bóng drawShadow(ctx, x + w / 2, y + h); // Chân ctx.fillStyle = "#171717"; drawRoundRect( ctx, x + 37, y + h - 48, 20, 48, 8 ); drawRoundRect( ctx, x + 66, y + h - 48, 20, 48, 8 ); // Giày ctx.fillStyle = "#111827"; drawRoundRect( ctx, x + 28, y + h - 15, 31, 13, 7 ); drawRoundRect( ctx, x + 65, y + h - 15, 31, 13, 7 ); // Thân hình 6 múi ctx.fillStyle = "#f59e0b"; drawRoundRect( ctx, x + 30, y + 65, 65, 70, 22 ); // Áo/đai ctx.fillStyle = "#dc2626"; ctx.fillRect( x + 31, y + 112, 63, 12 ); // Cơ bụng ctx.strokeStyle = "#92400e"; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(x + 62, y + 78); ctx.lineTo(x + 62, y + 112); ctx.moveTo(x + 45, y + 87); ctx.lineTo(x + 56, y + 87); ctx.moveTo(x + 68, y + 87); ctx.lineTo(x + 79, y + 87); ctx.moveTo(x + 45, y + 101); ctx.lineTo(x + 56, y + 101); ctx.moveTo(x + 68, y + 101); ctx.lineTo(x + 79, y + 101); ctx.stroke(); // Đầu ctx.fillStyle = "#f5b97a"; ctx.beginPath(); ctx.arc( x + 62, y + 42, 28, 0, Math.PI * 2 ); ctx.fill(); // Tóc ctx.fillStyle = "#111827"; ctx.beginPath(); ctx.arc( x + 62, y + 31, 27, Math.PI, Math.PI * 2 ); ctx.fill(); // Mắt ctx.fillStyle = "#111"; ctx.fillRect(x + 72, y + 39, 5, 5); // Tay ctx.strokeStyle = "#f5b97a"; ctx.lineWidth = 15; ctx.lineCap = "round"; if (attacking) { // Đấm về phía trước ctx.beginPath(); ctx.moveTo(x + 35, y + 82); ctx.lineTo(x + 100 + attackMove, y + 72); ctx.stroke(); // Nắm đấm ctx.fillStyle = "#f5b97a"; ctx.beginPath(); ctx.arc( x + 103 + attackMove, y + 70, 11, 0, Math.PI * 2 ); ctx.fill(); } else { ctx.beginPath(); ctx.moveTo(x + 35, y + 80); ctx.lineTo(x + 20, y + 115); ctx.stroke(); ctx.beginPath(); ctx.moveTo(x + 90, y + 80); ctx.lineTo(x + 105, y + 115); ctx.stroke(); } // Aura skill if (state === "skill1" || state === "skill2" || state === "skill3") { ctx.strokeStyle = state === "skill1" ? "#38bdf8" : state === "skill2" ? "#facc15" : "#ef4444"; ctx.lineWidth = 5; ctx.beginPath(); ctx.arc( x + 62, y + 82, 55 + Math.sin(t) * 5, 0, Math.PI * 2 ); ctx.stroke(); } } // ======================================================== // KIẾM SĨ // ======================================================== else if (hero === "kiemsi") { drawShadow(ctx, x + w / 2, y + h); // Áo choàng ctx.fillStyle = "#374151"; ctx.beginPath(); ctx.moveTo(x + 28, y + 70); ctx.lineTo(x + 96, y + 70); ctx.lineTo(x + 108, y + h - 20); ctx.lineTo(x + 20, y + h - 20); ctx.closePath(); ctx.fill(); // Giáp ctx.fillStyle = "#9ca3af"; drawRoundRect( ctx, x + 35, y + 65, 55, 65, 10 ); // Chân ctx.fillStyle = "#1f2937"; drawRoundRect( ctx, x + 37, y + 118, 20, 50, 5 ); drawRoundRect( ctx, x + 66, y + 118, 20, 50, 5 ); // Đầu ctx.fillStyle = "#eab27b"; ctx.beginPath(); ctx.arc( x + 62, y + 40, 26, 0, Math.PI * 2 ); ctx.fill(); // Tóc ctx.fillStyle = "#3f2418"; ctx.beginPath(); ctx.arc( x + 62, y + 32, 26, Math.PI, Math.PI * 2 ); ctx.fill(); // Mắt ctx.fillStyle = "#111"; ctx.fillRect( x + 72, y + 40, 5, 5 ); // Kiếm ctx.save(); ctx.translate( x + 92, y + 105 ); if (attacking) { ctx.rotate(-0.65); } else { ctx.rotate(-0.25); } // Lưỡi kiếm ctx.fillStyle = "#e5e7eb"; ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(90, -8); ctx.lineTo(100, 0); ctx.lineTo(90, 8); ctx.closePath(); ctx.fill(); // Chuôi ctx.fillStyle = "#92400e"; ctx.fillRect( -25, -4, 28, 8 ); // Tsuba ctx.fillStyle = "#facc15"; ctx.fillRect( -5, -12, 8, 24 ); ctx.restore(); // Skill aura if (state === "skill1" || state === "skill2" || state === "skill3") { ctx.strokeStyle = state === "skill1" ? "#60a5fa" : state === "skill2" ? "#a855f7" : "#facc15"; ctx.lineWidth = 5; ctx.beginPath(); ctx.arc( x + 62, y + 85, 60, -1, 1 ); ctx.stroke(); } } // ======================================================== // PHÁP SƯ // ======================================================== else if (hero === "phapsu") { drawShadow(ctx, x + w / 2, y + h); // Áo choàng ctx.fillStyle = "#581c87"; ctx.beginPath(); ctx.moveTo(x + 25, y + 65); ctx.lineTo(x + 100, y + 65); ctx.lineTo(x + 110, y + h); ctx.lineTo(x + 15, y + h); ctx.closePath(); ctx.fill(); // Viền áo ctx.strokeStyle = "#a855f7"; ctx.lineWidth = 4; ctx.stroke(); // Chân ctx.fillStyle = "#111827"; drawRoundRect( ctx, x + 40, y + h - 25, 18, 25, 6 ); drawRoundRect( ctx, x + 67, y + h - 25, 18, 25, 6 ); // Đầu ctx.fillStyle = "#f2c094"; ctx.beginPath(); ctx.arc( x + 62, y + 43, 25, 0, Math.PI * 2 ); ctx.fill(); // Tóc ctx.fillStyle = "#111827"; ctx.beginPath(); ctx.arc( x + 62, y + 32, 27, Math.PI, Math.PI * 2 ); ctx.fill(); // Mắt ctx.fillStyle = "#111"; ctx.fillRect( x + 72, y + 40, 5, 5 ); // Mũ pháp sư ctx.fillStyle = "#312e81"; ctx.beginPath(); ctx.moveTo(x + 27, y + 25); ctx.lineTo(x + 62, y - 20); ctx.lineTo(x + 97, y + 25); ctx.closePath(); ctx.fill(); ctx.fillStyle = "#facc15"; ctx.beginPath(); ctx.arc( x + 62, y + 12, 5, 0, Math.PI * 2 ); ctx.fill(); // Gậy phép ctx.strokeStyle = "#78350f"; ctx.lineWidth = 7; ctx.beginPath(); ctx.moveTo( x + 95, y + 155 ); ctx.lineTo( x + 108, y + 62 ); ctx.stroke(); // Quả cầu phép ctx.fillStyle = "#38bdf8"; ctx.shadowColor = "#38bdf8"; ctx.shadowBlur = 18; ctx.beginPath(); ctx.arc( x + 108, y + 55, 12 + Math.sin(t) * 3, 0, Math.PI * 2 ); ctx.fill(); ctx.shadowBlur = 0; // Phép thuật if (state === "skill1" || state === "skill2" || state === "skill3") { const color = state === "skill1" ? "#38bdf8" : state === "skill2" ? "#a855f7" : "#facc15"; ctx.strokeStyle = color; ctx.lineWidth = 4; for (let i = 0; i < 6; i++) { const angle = t + i * Math.PI / 3; const radius = 55; ctx.beginPath(); ctx.arc( x + 62 + Math.cos(angle) * radius, y + 85 + Math.sin(angle) * radius, 7, 0, Math.PI * 2 ); ctx.stroke(); } } } // ======================================================== // CUNG THỦ // ======================================================== else if (hero === "cungthu") { drawShadow(ctx, x + w / 2, y + h); // Áo xanh lá ctx.fillStyle = "#166534"; drawRoundRect( ctx, x + 30, y + 65, 65, 70, 15 ); // Quần ctx.fillStyle = "#1f2937"; drawRoundRect( ctx, x + 37, y + 125, 20, 43, 6 ); drawRoundRect( ctx, x + 66, y + 125, 20, 43, 6 ); // Đầu ctx.fillStyle = "#f3c18b"; ctx.beginPath(); ctx.arc( x + 62, y + 42, 25, 0, Math.PI * 2 ); ctx.fill(); // Tóc vàng ctx.fillStyle = "#facc15"; ctx.beginPath(); ctx.arc( x + 62, y + 33, 28, Math.PI, Math.PI * 2 ); ctx.fill(); // Tóc bên ctx.fillRect( x + 35, y + 28, 10, 35 ); ctx.fillRect( x + 79, y + 28, 10, 35 ); // Mắt ctx.fillStyle = "#111"; ctx.fillRect( x + 72, y + 40, 5, 5 ); // Cánh tay ctx.strokeStyle = "#f3c18b"; ctx.lineWidth = 12; ctx.lineCap = "round"; ctx.beginPath(); ctx.moveTo( x + 40, y + 80 ); ctx.lineTo( x + 95, y + 90 ); ctx.stroke(); // Cung ctx.strokeStyle = "#92400e"; ctx.lineWidth = 6; ctx.beginPath(); ctx.arc( x + 100, y + 100, 45, -Math.PI / 2, Math.PI / 2 ); ctx.stroke(); // Dây cung ctx.strokeStyle = "#fef3c7"; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo( x + 100, y + 55 ); ctx.lineTo( x + 100, y + 145 ); ctx.stroke(); // Mũi tên ctx.strokeStyle = "#d1d5db"; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo( x + 55, y + 100 ); ctx.lineTo( x + 135, y + 100 ); ctx.stroke(); // Đầu mũi tên ctx.fillStyle = "#e5e7eb"; ctx.beginPath(); ctx.moveTo( x + 140, y + 100 ); ctx.lineTo( x + 130, y + 94 ); ctx.lineTo( x + 130, y + 106 ); ctx.closePath(); ctx.fill(); // Khi tấn công: mũi tên bay if (attacking) { ctx.strokeStyle = "#facc15"; ctx.lineWidth = 4; ctx.beginPath(); ctx.moveTo( x + 100, y + 90 ); ctx.lineTo( x + 145 + attackMove, y + 90 ); ctx.stroke(); } // Skill aura if (state === "skill1" || state === "skill2" || state === "skill3") { ctx.strokeStyle = state === "skill1" ? "#22c55e" : state === "skill2" ? "#38bdf8" : "#facc15"; ctx.lineWidth = 4; ctx.beginPath(); ctx.arc( x + 62, y + 85, 58 + Math.sin(t) * 4, 0, Math.PI * 2 ); ctx.stroke(); } } ctx.restore(); } // ============================================================ // HÌNH CHỮ NHẬT BO GÓC // ============================================================ function drawRoundRect(ctx, x, y, width, height, radius) { ctx.beginPath(); ctx.moveTo(x + radius, y); ctx.lineTo(x + width - radius, y); ctx.quadraticCurveTo( x + width, y, x + width, y + radius ); ctx.lineTo( x + width, y + height - radius ); ctx.quadraticCurveTo( x + width, y + height, x + width - radius, y + height ); ctx.lineTo( x + radius, y + height ); ctx.quadraticCurveTo( x, y + height, x, y + height - radius ); ctx.lineTo( x, y + radius ); ctx.quadraticCurveTo( x, y, x + radius, y ); ctx.closePath(); ctx.fill(); } // ============================================================ // BÓNG DƯỚI CHÂN // ============================================================ function drawShadow(ctx, x, y) { ctx.save(); ctx.fillStyle = "rgba(0,0,0,0.35)"; ctx.beginPath(); ctx.ellipse( x, y, 35, 8, 0, 0, Math.PI * 2 ); ctx.fill(); ctx.restore(); }
